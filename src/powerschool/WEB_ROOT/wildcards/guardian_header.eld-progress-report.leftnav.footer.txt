<script type="text/javascript">
// Get ELD status using tlist_sql
let eldStatus = '';
~[tlist_sql;
SELECT COALESCE(S_PA_STU_X.LEP_ELL_Status_Code, '00') as eld_status
FROM students 
LEFT JOIN S_PA_STU_X ON S_PA_STU_X.studentsdcid = students.dcid
WHERE students.id = ~(curstudid)]eldStatus = '~(eld_status)';[/tlist_sql]

document.addEventListener('DOMContentLoaded', function() {
    // Only show for ELD students (LEP_ELL_Status_Code = '01')
    if (eldStatus === '01') {
        const menuContainer = document.querySelector('[role="menu"]');
        if (menuContainer) {
            const eldLink = '<li id="btn-eldprogress" role="menuitem"><a href="/guardian/eld-progress-report/dashboard.html">ELD Progress Report</a></li>';
            menuContainer.insertAdjacentHTML('afterbegin', eldLink);
        }
    }
});
</script>
